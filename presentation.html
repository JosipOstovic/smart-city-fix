Why this shape? Photos never travel through Express — the React client uploads them directly to Cloudinary using an unsigned preset and only sends the resulting URL string to the backend. That keeps Node out of the binary-streaming business and dramatically reduces server load.

4 · Backend bootstrap — app.js

Where every request enters and gets routed.

backend/src/app.js 
const express      = require('express');
const cors         = require('cors');
const cookieParser = require('cookie-parser');
const errorHandler = require('./middleware/errorHandler');

const app = express();

app.use(cors({
  origin: process.env.CORS_ORIGIN || 'http://localhost:5173',
  credentials: true,                 // ← must be true so the JWT cookie crosses origins
}));
app.use(cookieParser());
app.use(express.json({ limit: '10mb' }));

// Routes
app.use('/api/auth',             require('./routes/auth'));
app.use('/api/issues',           require('./routes/issues'));
app.use('/api/categories',       require('./routes/categories'));
app.use('/api/admin/statistics', require('./routes/statistics'));
app.use('/api/admin',            require('./routes/admin'));
app.use('/api/users',            require('./routes/users'));
app.use('/api',                  require('./routes/comments'));
app.use('/api/notifications',    require('./routes/notifications'));

app.use(errorHandler);   // must be last so it catches errors from any route
module.exports = app;
Order matters. /api/admin/statistics is mounted before /api/admin — otherwise the broader /api/admin router would swallow the request first. The same trick is used in routes/issues.js, where /stats/public is declared before /:id.

5 · Authentication — JWT in an httpOnly cookie

No tokens in localStorage. The browser carries the cookie automatically.

Login produces a signed JWT

backend/src/services/authService.js 
async function login(email, password) {
  const user = await User.findByEmail(email);
  if (!user) throw { status: 401, message: 'Invalid email or password' };

  const isMatch = await bcrypt.compare(password, user.password_hash);
  if (!isMatch) throw { status: 401, message: 'Invalid email or password' };

  const token = jwt.sign(
    { id: user.id, email: user.email, role: user.role },
    config.jwtSecret,
    { expiresIn: config.jwtExpiresIn }   // default 7d
  );
  return { user: sanitizeUser(user), token };
}

Route writes it as an httpOnly cookie

backend/src/routes/auth.js 
const COOKIE_OPTIONS = {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'lax',
  maxAge: 7 * 24 * 60 * 60 * 1000,

Two middleware flavors

backend/src/middleware/auth.js 
function authenticate(req, res, next) {        // hard gate: 401 if no/invalid token
  const token = req.cookies.token;
  if (!token) return res.status(401).json({ error: { code: 'UNAUTHORIZED' } });
  try {
    req.user = jwt.verify(token, config.jwtSecret);
    next();
  } catch { res.status(401).json({ error: { code: 'UNAUTHORIZED' } }); }
}

function optionalAuth(req, res, next) {       // soft: attach user if present, otherwise pass
  const token = req.cookies.token;
  if (token) { try { req.user = jwt.verify(token, config.jwtSecret); } catch {} }
  next();
}

Role gate for admin endpoints

backend/src/middleware/authorize.js · routes/admin.js 
function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role))
      return res.status(403).json({ error: { code: 'FORBIDDEN' } });
    next();
  };
}

// Every admin endpoint runs through both gates:
router.use(authenticate, requireRole('admin'));
Why httpOnly cookies? JS can't read them, so XSS can't steal the token. The cost is enabling CORS credentials: true + axios withCredentials: true so they actually travel.

Reverse-geocoding happens server-side

backend/src/utils/geocoding.js 
async function reverseGeocode(latitude, longitude) {
  try {
    const url = `https://maps.googleapis.com/maps/api/geocode/json`
      + `?latlng=${latitude},${longitude}&key=${config.googleMapsApiKey}&language=hr`;
    const response = await fetch(url);
    const data     = await response.json();
    if (data.status === 'OK' && data.results.length > 0)
      return data.results[0].formatted_address;
    return `${latitude.toFixed(6)}, ${longitude.toFixed(6)}`;  // graceful fallback
  } catch (error) { return `${latitude}, ${longitude}`; }
}
Why server-side? The Geocoding API key needs to stay private. Doing the lookup in issueService.create means the key never ships to the browser, and we get a stable, formatted address baked into the row at write time.

8 · Admin status changes & the notification trigger

One PATCH does three things atomically: update issue, append history, notify owner.

backend/src/services/adminService.js 
async changeStatus(issueId, { status, note }, adminId) {
  const issue = await db('issues').where('id', issueId).first();
  if (!issue) throw Object.assign(new Error('Issue not found'), { status: 404 });

  const oldStatus = issue.status;
  await db('issues').where('id', issueId).update({ status, updated_at: new Date() });

  const historyEntry = await StatusHistory.create({
    issue_id: issueId, old_status: oldStatus, new_status: status,
    changed_by: adminId, note: note || null,
  });

  // Build a Croatian-localized message and store it for the citizen
  const statusLabels = { reported: 'Prijavljeno', in_progress: 'U tijeku',
                        resolved: 'Riješeno', rejected: 'Odbijeno' };
  await Notification.create({
    user_id:  issue.user_id,
    issue_id: issueId,
    type:     'status_change',
    message:  `Status vašeg problema "${issue.title}" promijenjen u: ${statusLabels[status]}`,
  });

  return { issue: { id: issueId, status }, history_entry: historyEntry };
}

Citizen polls for the new notification

frontend/src/context/NotificationContext.jsx 
useEffect(() => {
  fetchNotifications();
  const interval = setInterval(fetchNotifications, 60000);  // poll every 60s
  return () => clearInterval(interval);
}, [fetchNotifications]);

const markAsRead = async (id) => {
  await api.patch(`/notifications/${id}/read`);
  setNotifications(prev => prev.map(n => n.id === id ? { ...n, is_read: true } : n));
  setUnreadCount(prev => Math.max(0, prev - 1));
};
Polling, not WebSockets. The 60-second interval is a deliberately simple choice for this app's traffic level — it costs one cheap query per minute per active citizen and avoids the operational complexity of long-lived connections.

export function AuthProvider({ children }) {
  const [user, setUser]       = useState(null);
  const [loading, setLoading] = useState(true);

  // On every full page load, ask the backend who we are (cookie travels automatically).
  useEffect(() => {
    api.get('/auth/me')
      .then(res  => setUser(res.data.user))
      .catch(()  => setUser(null))
      .finally(()=> setLoading(false));
  }, []);

  const login    = async (email, password) => { ... setUser(res.data.user); };
  const register = async (data)            => { ... };
  const logout   = async ()                => { await api.post('/auth/logout'); setUser(null); };

  return <AuthContext.Provider value={{ user, loading, login, register, logout }}>{children}</AuthContext.Provider>;
}

export default function AdminRoute({ children }) {

11 · Photo upload — Cloudinary direct upload

The browser talks to Cloudinary itself; the backend never sees the bytes.

frontend/src/services/cloudinaryService.js 
const CLOUD_NAME    = import.meta.env.VITE_CLOUDINARY_CLOUD_NAME;
const UPLOAD_PRESET = import.meta.env.VITE_CLOUDINARY_UPLOAD_PRESET;

export async function uploadImage(file) {
  const formData = new FormData();
  formData.append('file', file);
  formData.append('upload_preset', UPLOAD_PRESET);   // unsigned preset

  const res = await fetch(
    `https://api.cloudinary.com/v1_1/${CLOUD_NAME}/image/upload`,
    { method: 'POST', body: formData },
  );
  if (!res.ok) throw new Error('Image upload failed');
  return (await res.json()).secure_url;     // just the URL, that's all the API needs
}

12 · Admin statistics — one query, four insights

Aggregations done in PostgreSQL, rendered with Recharts.

backend/src/services/statisticsService.js (excerpt) 
// 1) Counts by status
const byStatus = await db('issues').select('status').count('id as count').groupBy('status');

// 2) Counts by category — joined for human-readable names
const byCategory = await db('issues')
  .join('categories', 'issues.category_id', 'categories.id')
  .select('categories.name as category_name')
  .count('issues.id as count')
  .groupBy('categories.name')
  .orderBy('count', 'desc');

// 3) Monthly trend, last 6 months
const monthlyTrend = await db('issues')
  .select(db.raw("to_char(created_at, 'YYYY-MM') as month"))
  .count('id as count')
  .where('created_at', '>=', sixMonthsAgo)
  .groupByRaw("to_char(created_at, 'YYYY-MM')");

// 4) Average resolution time in hours, joining status_history
const avgResolution = await db('status_history')
  .join('issues', 'status_history.issue_id', 'issues.id')
  .where('status_history.new_status', 'resolved')
  .select(db.raw(
    "avg(extract(epoch from (status_history.created_at - issues.created_at)) / 3600) as avg_hours"
  )).first();

The whole app, in one sentence. A React SPA reads and writes a single Express API guarded by JWT cookies; that API uses Knex to talk to PostgreSQL, calls Google for reverse geocoding, lets the browser upload photos straight to Cloudinary, and records every admin status change as both a history row and a citizen-facing notification.
